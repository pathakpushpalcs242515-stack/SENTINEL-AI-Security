def normalize_url(raw_url):
    if raw_url is None:
        raise ValueError("A URL is required.")

    candidate = str(raw_url).strip()
    if not candidate:
        raise ValueError("A URL is required.")

    if "://" not in candidate:
        candidate = f"https://{candidate}"

    parsed = urlsplit(candidate)
    if parsed.scheme.lower() not in {"http", "https"} or not parsed.netloc:
        raise ValueError("Please provide a valid HTTP or HTTPS URL.")

    return parsed.geturl()


def score_to_risk(score):
    if score >= 70:
        return "HIGH"
    if score >= 30:
        return "MODERATE"
    return "LOW"


def is_clean_public_domain(hostname: str) -> bool:
    if not hostname:
        return False
    labels = [part for part in hostname.split(".") if part]
    if len(labels) < 2:
        return False
    if hostname.startswith("xn--"):
        return False
    if any(part in {"localhost", "local", "internal", "test", "example"} for part in labels):
        return True
    if any(token in hostname for token in ["login", "signin", "verify", "secure", "update", "payment", "confirm", "account", "bank", "wallet", "recover", "suspended", "credential"]):
        return False
    if hostname.count("-") >= 2 or len(hostname) > 45 or hostname.count(".") > 3:
        return False
    return True


def verify_real_url(url: str):
    parsed = urlsplit(url)
    hostname = (parsed.hostname or "").lower()
    result = {
        "dnsResolved": False,
        "reachable": False,
        "tlsValid": False,
        "httpStatus": None,
        "redirectedTo": None,
        "finalUrl": url,
        "error": None,
    }

    if not hostname:
        result["error"] = "No hostname was found in the provided URL."
        return result

    try:
        socket.getaddrinfo(hostname, parsed.port or (443 if parsed.scheme.lower() == "https" else 80), type=socket.SOCK_STREAM)
        result["dnsResolved"] = True
    except socket.gaierror as exc:
        result["error"] = f"DNS resolution failed: {exc}"
        return result

    try:
        ctx = ssl.create_default_context()
        req = urllib_request.Request(url, method="HEAD")
        with urllib_request.urlopen(req, timeout=5, context=ctx) as response:
            result["reachable"] = True
            result["httpStatus"] = getattr(response, "status", None)
            result["finalUrl"] = response.geturl() or url
            result["redirectedTo"] = response.geturl() if response.geturl() != url else None
            result["tlsValid"] = True
    except urllib_error.HTTPError as exc:
        result["reachable"] = True
        result["httpStatus"] = getattr(exc, "code", None)
        result["finalUrl"] = exc.geturl() or url
        result["redirectedTo"] = exc.geturl() if exc.geturl() and exc.geturl() != url else None
        result["tlsValid"] = True
    except Exception:
        try:
            req = urllib_request.Request(url, method="GET")
            with urllib_request.urlopen(req, timeout=5, context=ssl.create_default_context()) as response:
                result["reachable"] = True
                result["httpStatus"] = getattr(response, "status", None)
                result["finalUrl"] = response.geturl() or url
                result["redirectedTo"] = response.geturl() if response.geturl() != url else None
                result["tlsValid"] = True
        except urllib_error.HTTPError as exc:
            result["reachable"] = True
            result["httpStatus"] = getattr(exc, "code", None)
            result["finalUrl"] = exc.geturl() or url
            result["redirectedTo"] = exc.geturl() if exc.geturl() and exc.geturl() != url else None
            result["tlsValid"] = True
        except Exception as exc:
            result["error"] = str(exc)
            if parsed.scheme.lower() == "https":
                result["tlsValid"] = False

    return result


def build_scan_result(url):
    if MODEL_BUNDLE is None:
        raise RuntimeError(MODEL_LOAD_ERROR or "The phishing model is not available.")

    feature_values = extract_features(url)
    feature_row = pd.DataFrame([feature_values], columns=FEATURE_NAMES)
    probabilities = MODEL_BUNDLE.predict_proba(feature_row)[0]
    classes = list(getattr(MODEL_BUNDLE, "classes_", [0, 1]))

    phishing_index = 1 if 1 in classes else (len(classes) - 1)
    phishing_probability = float(probabilities[phishing_index])
    parsed = urlsplit(url)
    hostname = (parsed.hostname or "unknown").lower()
    reasons = explain_features(url)
    real_check = verify_real_url(url)

    risk_score = max(0, min(100, round(phishing_probability * 100)))
    if real_check.get("dnsResolved") and real_check.get("reachable") and real_check.get("tlsValid"):
        risk_score = max(0, risk_score - 35)
    if is_clean_public_domain(hostname) and not any(token in hostname for token in ["login", "signin", "verify", "secure", "update", "payment", "confirm", "account", "bank", "wallet", "recover", "suspended", "credential"]):
        risk_score = min(risk_score, 25)
    if not real_check.get("dnsResolved") or not real_check.get("reachable"):
        risk_score = max(risk_score, 60)

    confidence = max(0, min(100, round(float(max(probabilities)) * 100)))

    if risk_score >= 70:
        verdict = "This link looks highly suspicious and is likely a phishing or malicious destination."
    elif risk_score >= 30:
        verdict = "This link shows moderate risk indicators, so it should be checked carefully before login or payment actions."
    else:
        verdict = "This link does not show strong phishing indicators and the domain appears to be reachable and legitimate on basic inspection."

    return {
        "url": url,
        "hostname": hostname,
        "riskScore": risk_score,
        "riskLevel": score_to_risk(risk_score),
        "confidence": confidence,
        "model": "Random Forest",
        "verdict": verdict,
        "reasons": reasons,
        "realCheck": real_check,
    }

@app.route("/api/scan", methods=["POST"])
def scan_url():
    payload = request.get_json(silent=True) or {}
    raw_url = payload.get("url") or request.form.get("url")

    try:
        url = normalize_url(raw_url)
    except ValueError as exc:
        return jsonify({"error": str(exc)}), 400

    try:
        result = build_scan_result(url)
    except Exception as exc:  # pragma: no cover - model availability is runtime checked
        return jsonify({"error": str(exc)}), 503

    user_id = request.headers.get("X-Sentinel-User") or "anonymous"
    reasons_json = json.dumps(result["reasons"], ensure_ascii=False)

    conn = get_db_connection()
    conn.execute(
        """
        INSERT INTO scans (user_id, url, hostname, risk_level, risk_score, confidence, verdict, model, reasons)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        """,
        (
            user_id,
            url,
            result["hostname"],
            result["riskLevel"],
            result["riskScore"],
            result["confidence"],
            result["verdict"],
            result["model"],
            reasons_json,
        ),
    )
    conn.commit()
    last_id = conn.execute("SELECT last_insert_rowid() AS id").fetchone()["id"]
    conn.close()

    result["id"] = last_id
    result["createdAt"] = __import__("datetime").datetime.utcnow().strftime("%Y-%m-%d %H:%M:%S")
    return jsonify({"scan": result, "status": "ok"})

"""Lexical URL feature extraction shared by training and inference."""
from collections import Counter
from math import log2
from urllib.parse import urlsplit
import ipaddress
import re

SHORTENERS = {
    "bit.ly", "tinyurl.com", "t.co", "is.gd", "ow.ly", "buff.ly",
    "cutt.ly", "rb.gy", "shorturl.at", "rebrand.ly", "tiny.cc", "lnkd.in"
}
SUSPICIOUS_TERMS = (
    "login", "signin", "verify", "secure", "account", "update", "password",
    "wallet", "gift", "payment", "urgent", "confirm", "auth", "banking",
    "recover", "suspended", "webscr", "credential", "invoice", "support"
)
FEATURE_NAMES = [
    "url_length", "hostname_length", "path_length", "query_length", "tld_length",
    "count_dots", "count_hyphens", "count_underscores", "count_slashes", "count_at",
    "count_question", "count_equal", "count_ampersand", "count_percent", "count_digits",
    "digit_ratio", "count_subdomains", "has_ip", "uses_https", "has_punycode",
    "has_port", "has_userinfo", "suspicious_keyword_count", "is_shortener",
    "hostname_entropy", "has_double_slash_path", "count_letters", "special_char_ratio"
]

def extract_features(raw_url: str) -> dict:
    raw = str(raw_url).strip()
    candidate = raw if "://" in raw else "http://" + raw
    try:
        parsed = urlsplit(candidate)
        host = (parsed.hostname or "").lower().strip(".")
        path = parsed.path or ""
        query = parsed.query or ""
        port = parsed.port
        userinfo = bool(parsed.username or parsed.password)
    except (ValueError, UnicodeError):
        host, path, query, port, userinfo = "", "", "", None, False
        parsed = urlsplit("http://invalid.invalid")
    labels = [part for part in host.split(".") if part]
    try:
        ipaddress.ip_address(host)
        is_ip = 1
    except ValueError:
        is_ip = 0
    counts = Counter(raw.lower())
    letters = sum(ch.isalpha() for ch in raw)
    digits = sum(ch.isdigit() for ch in raw)
    specials = sum(not ch.isalnum() for ch in raw)
    freq = Counter(host)
    entropy = -sum((n / max(1, len(host))) * log2(n / max(1, len(host))) for n in freq.values()) if host else 0.0
    terms = sum(1 for term in SUSPICIOUS_TERMS if term in raw.lower())
    tld = labels[-1] if len(labels) > 1 else ""
    path_after_host = path.lstrip("/")
    return {
        "url_length": len(raw), "hostname_length": len(host), "path_length": len(path),
        "query_length": len(query), "tld_length": len(tld), "count_dots": counts["."],
        "count_hyphens": counts["-"], "count_underscores": counts["_"],
        "count_slashes": counts["/"], "count_at": counts["@"], "count_question": counts["?"],
        "count_equal": counts["="], "count_ampersand": counts["&"], "count_percent": counts["%"],
        "count_digits": digits, "digit_ratio": digits / max(1, len(raw)),
        "count_subdomains": max(0, len(labels) - 2), "has_ip": is_ip,
        "uses_https": int(parsed.scheme.lower() == "https"),
        "has_punycode": int("xn--" in host), "has_port": int(port is not None),
        "has_userinfo": int(userinfo), "suspicious_keyword_count": terms,
        "is_shortener": int(host in SHORTENERS), "hostname_entropy": entropy,
        "has_double_slash_path": int("//" in path_after_host), "count_letters": letters,
        "special_char_ratio": specials / max(1, len(raw)),
    }

def explain_features(raw_url: str) -> list[str]:
    f = extract_features(raw_url)
    candidate = raw_url if "://" in raw_url else "http://" + raw_url
    parsed = urlsplit(candidate)
    host = (parsed.hostname or "").lower()
    reasons = []
    if not f["uses_https"]: reasons.append("The URL does not use HTTPS.")
    if f["has_ip"]: reasons.append("The hostname is a raw IP address rather than a domain name.")
    if f["has_punycode"]: reasons.append("The domain contains punycode, which can represent look-alike characters.")
    if f["count_subdomains"] >= 3: reasons.append("The hostname contains several subdomain levels.")
    if f["has_userinfo"]: reasons.append("The URL contains embedded username or password information.")
    if f["is_shortener"]: reasons.append("The domain is a URL shortener, so the destination is not obvious from the link.")
    terms = [term for term in SUSPICIOUS_TERMS if term in raw_url.lower()]
    if terms: reasons.append("The URL contains potentially sensitive or urgency-related terms: " + ", ".join(terms[:5]) + ".")
    if len(raw_url) > 180: reasons.append("The URL is unusually long.")
    if host.count("-") >= 3: reasons.append("The hostname contains several hyphens.")
    if not reasons: reasons.append("No obvious lexical warning signs were detected; this does not prove the destination is safe.")
    return reasons

"""Download the public UCI PhiUSIIL dataset, train and evaluate a URL-only classifier.

UCI dataset: https://archive.ics.uci.edu/dataset/967/phiusiil+phishing+url+dataset
Target definition in UCI: 1 = legitimate, 0 = phishing.
This script uses only the raw URL column and recomputes lexical features to avoid
relying on precomputed dataset features at inference time.
"""
import json
import os
from pathlib import Path
import sys

import joblib
import pandas as pd
from sklearn.ensemble import RandomForestClassifier, ExtraTreesClassifier, VotingClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, confusion_matrix, roc_auc_score
from sklearn.model_selection import train_test_split

HERE = Path(__file__).resolve().parent
MODEL_DIR = HERE / "models"
MODEL_DIR.mkdir(parents=True, exist_ok=True)
sys.path.insert(0, str(HERE))
from features import FEATURE_NAMES, extract_features

MAX_ROWS = int(os.getenv("SENTINEL_TRAIN_ROWS", "50000"))
RANDOM_STATE = 42

def load_dataset():
    local = os.getenv("SENTINEL_DATASET_CSV")
    if local and Path(local).exists():
        df = pd.read_csv(local)
        url_col = next((c for c in df.columns if c.lower() == "url"), None)
        label_col = next((c for c in df.columns if c.lower() in {"label", "status", "result", "target"}), None)
        if not url_col or not label_col:
            raise ValueError("Local CSV must contain URL and label columns.")
        return df[[url_col, label_col]].rename(columns={url_col: "URL", label_col: "label"})
    from ucimlrepo import fetch_ucirepo
    print("Downloading PhiUSIIL Phishing URL dataset from UCI…", flush=True)
    dataset = fetch_ucirepo(id=967)
    X, y = dataset.data.features, dataset.data.targets
    url_col = next((c for c in X.columns if c.lower() == "url"), None)
    if not url_col:
        raise ValueError("The UCI dataset did not expose a raw URL column.")
    target_col = y.columns[0]
    return pd.DataFrame({"URL": X[url_col].astype(str), "label": y[target_col]})

def normalize_phishing_label(series):
    # UCI PhiUSIIL labels: 1 legitimate, 0 phishing. Other CSV labels are accepted
    # if they are recognizable; numeric binary CSV labels default to 1=phishing.
    vals = set(series.dropna().astype(str).str.strip().str.lower().unique())
    if vals and vals.issubset({"0", "1", "0.0", "1.0"}):
        return (pd.to_numeric(series) == 0).astype(int)
    mapping = {"phishing": 1, "malicious": 1, "bad": 1, "unsafe": 1,
               "legitimate": 0, "benign": 0, "good": 0, "safe": 0,
               "phish": 1, "1": 1, "0": 0}
    normalized = series.astype(str).str.strip().str.lower().map(mapping)
    if normalized.isna().any():
        raise ValueError("Labels must identify phishing/malicious vs legitimate/benign URLs.")
    return normalized.astype(int)

def main():
    df = load_dataset().dropna(subset=["URL", "label"]).copy()
    df["URL"] = df["URL"].astype(str).str.strip()
    df = df[df["URL"].str.len().between(4, 2048)].drop_duplicates(subset=["URL"])
    df["target"] = normalize_phishing_label(df["label"])
    if len(df) > MAX_ROWS:
        # Stratified sample preserves both classes while keeping deployment build time reasonable.
        df, _ = train_test_split(df, train_size=MAX_ROWS, random_state=RANDOM_STATE, stratify=df["target"])
    print(f"Training on {len(df):,} unique URLs; phishing={int(df.target.sum()):,}, legitimate={int((1-df.target).sum()):,}", flush=True)
    feature_rows = [extract_features(url) for url in df["URL"].tolist()]
    X = pd.DataFrame(feature_rows, columns=FEATURE_NAMES).fillna(0)
    y = df["target"].astype(int)
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_STATE, stratify=y)
    model = RandomForestClassifier(n_estimators=120, max_depth=24, min_samples_leaf=2,
        class_weight="balanced_subsample", n_jobs=2, random_state=RANDOM_STATE)
    model.fit(X_train, y_train)
    pred = model.predict(X_test)
    proba = model.predict_proba(X_test)[:, 1]
    metrics = {
        "dataset": "UCI PhiUSIIL Phishing URL Dataset (URL-only re-engineered features)",
        "datasetSource": "https://archive.ics.uci.edu/dataset/967/phiusiil+phishing+url+dataset",
        "rowsUsed": int(len(df)), "phishingRows": int(y.sum()), "legitimateRows": int((1-y).sum()),
        "trainRows": int(len(X_train)), "testRows": int(len(X_test)),
        "accuracy": round(float(accuracy_score(y_test, pred)), 4),
        "precision": round(float(precision_score(y_test, pred, zero_division=0)), 4),
        "recall": round(float(recall_score(y_test, pred, zero_division=0)), 4),
        "f1": round(float(f1_score(y_test, pred, zero_division=0)), 4),
        "rocAuc": round(float(roc_auc_score(y_test, proba)), 4),
        "confusionMatrix": confusion_matrix(y_test, pred).tolist(),
        "featureCount": len(FEATURE_NAMES), "model": "Random Forest", "randomState": RANDOM_STATE,
    }
    joblib.dump({"model": model, "feature_names": FEATURE_NAMES, "metrics": metrics}, MODEL_DIR / "sentinel_url_model.joblib", compress=3)
    (MODEL_DIR / "metrics.json").write_text(json.dumps(metrics, indent=2), encoding="utf-8")
    print(json.dumps(metrics, indent=2), flush=True)
    print(f"Saved model to {MODEL_DIR / 'sentinel_url_model.joblib'}", flush=True)

function useScan() {
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState(null);
  const [error, setError] = useState('');
  const requestId = useRef(0);

  const analyzeUrl = useCallback(async (url) => {
    const currentRequestId = ++requestId.current;
    setLoading(true);
    setError('');
    setResult(null);
    try {
      const response = await fetch(`${API}/api/scan`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', ...userHeaders() },
        body: JSON.stringify({ url }),
      });
      const data = await response.json();
      if (!response.ok) throw new Error(data.error || 'Analysis failed.');
      if (!data.scan?.realCheck) {
        throw new Error('The backend is outdated. Restart the SENTINEL API, then scan again.');
      }
      if (requestId.current === currentRequestId) setResult(data.scan);
    } catch (e) {
      if (requestId.current === currentRequestId) {
        setError(e.message || 'Could not connect to the SENTINEL API.');
      }
    } finally {
      if (requestId.current === currentRequestId) setLoading(false);
    }
  }, []);

  const reset = useCallback(() => {
    requestId.current += 1;
    setLoading(false);
    setResult(null);
    setError('');
  }, []);

  return { loading, result, error, analyzeUrl, reset };
}

import { useEffect, useRef, useState } from 'react';
import { ScanSearch, ArrowRight, Camera, CameraOff } from 'lucide-react';
import { Html5Qrcode } from 'html5-qrcode';

export default function ScanForm({ onScan, loading = false }) {
  const [url, setUrl] = useState('');
  const [cameraOpen, setCameraOpen] = useState(false);
  const [cameraError, setCameraError] = useState('');
  const scannerRef = useRef(null);
  const processingRef = useRef(false);
  const onScanRef = useRef(onScan);

  useEffect(() => {
    onScanRef.current = onScan;
  }, [onScan]);

  useEffect(() => {
    if (!cameraOpen) return undefined;
    let cancelled = false;
    let scanner;

    async function start() {
      try {
        scanner = new Html5Qrcode('qr-reader');
        scannerRef.current = scanner;
        await scanner.start(
          { facingMode: 'environment' },
          { fps: 10, qrbox: { width: 220, height: 220 }, aspectRatio: 1 },
          async (decodedText) => {
            if (cancelled || processingRef.current) return;
            let parsed;
            try {
              parsed = new URL(decodedText.trim());
            } catch {
              setCameraError('This QR code does not contain a valid website URL.');
              return;
            }
            if (!['http:', 'https:'].includes(parsed.protocol)) {
              setCameraError('Please scan a QR code containing an HTTP or HTTPS URL.');
              return;
            }
            processingRef.current = true;
            setUrl(parsed.href);
            setCameraError('');
            setCameraOpen(false);
            onScanRef.current?.(parsed.href);
            processingRef.current = false;
          },
          () => {}
        );
      } catch {
        if (!cancelled) {
          setCameraError('Could not open the camera. Allow camera permission and try again.');
          setCameraOpen(false);
        }
      }
    }

    start();
    return () => {
      cancelled = true;
      if (scannerRef.current === scanner) scannerRef.current = null;
      if (scanner) {
        (async () => {
          try { if (scanner.isScanning) await scanner.stop(); } catch {}
          try { scanner.clear(); } catch {}
        })();
      }
    };
  }, [cameraOpen]);

  function submit(event) {
    event.preventDefault();
    setCameraError('');
    let parsed;
    try {
      parsed = new URL(url.trim());
    } catch {
      setCameraError('Enter a valid URL, such as https://example.com.');
      return;
    }
    if (!['http:', 'https:'].includes(parsed.protocol)) {
      setCameraError('Only HTTP and HTTPS website URLs are supported.');
      return;
    }
    onScanRef.current?.(parsed.href);
  }

  return (
    <section id="scanner" className="scanner-panel">
      <div className="scan-columns">
        <div className="qr-side">
          <div className="scanner-title"><span className="scanner-icon"><Camera size={21} /></span><h3>Scan a QR code</h3></div>
          <p>Use your camera to read a QR code containing a website URL.</p>
          {cameraOpen && <div className="qr-reader-container"><div id="qr-reader" /></div>}
          {cameraError && <p className="qr-error" role="alert">{cameraError}</p>}
          <button type="button" className="secondary-button" disabled={loading} onClick={() => { setCameraError(''); setCameraOpen((open) => !open); }}>
            {cameraOpen ? <CameraOff size={17} /> : <Camera size={17} />}
            {cameraOpen ? 'Stop camera' : 'Open QR camera'}
          </button>
        </div>
        <div className="url-side">
          <div className="scanner-title"><span className="scanner-icon"><ScanSearch size={21} /></span><h3>Analyze a URL</h3></div>
          <p>Paste a full URL to check for common phishing signals.</p>
          <form onSubmit={submit}>
            <label className="url-label" htmlFor="url-input">Website URL</label>
            <div className="url-input-row">
              <input id="url-input" type="url" value={url} onChange={(e) => { setUrl(e.target.value); setCameraError(''); }} placeholder="https://example.com/login" required disabled={loading} />
              <button className="primary-button" type="submit" disabled={loading || !url.trim()}>
                {loading ? 'Analyzing…' : <>Analyze <ArrowRight size={17} /></>}
              </button>
            </div>
          </form>
        </div>
      </div>
      <div className="scanner-footnote"><span>✓ URL checks only</span><span>✓ Does not open the scanned page</span><span>✓ Scan history saved to SQLite</span></div>
    </section>
  );
}
